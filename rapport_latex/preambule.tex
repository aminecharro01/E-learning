% --- Langue (français principal, anglais pour le bloc Abstract) ---
\usepackage[english,french]{babel}
\usepackage[T1]{fontenc}

% --- Police : rendu proche Times New Roman ---
\usepackage{mathptmx}

% --- Micro-typographie + tolérance de justification ---
% Nécessaire avec les marges resserrées du modèle (left=3cm) : de longs

% peuvent pas se couper et débordaient sinon visiblement dans la marge.

\setlength{\emergencystretch}{3em}

% --- Interligne 1,5 et retrait de première ligne ---
\usepackage{setspace}
\onehalfspacing

% --- Figures / tableaux ---

\graphicspath{{images/}{../diagrammes/images/}}
\usepackage{float}                 % positionnement stable avec [H]
\usepackage{rotating}              % figures en paysage (sidewaysfigure) pour les diagrammes très larges
\usepackage{tikz}                  % encadrés (fiche signalétique, schémas simples)
\usepackage[font=small,labelfont=bf,labelsep=period]{caption}
\captionsetup[table]{position=above}   % légendes tableaux au-dessus
\captionsetup[figure]{position=below}  % légendes figures en dessous

% Numérotation des figures et tableaux par chapitre → Figure 2.1, Tableau 3.2

\counterwithin{table}{chapter}

% --- Tableaux enrichis ---
\usepackage{booktabs}
\usepackage{longtable}
\usepackage{array}
\usepackage{multirow}
\usepackage{enumitem}

\usepackage{listings}

% --- Figure = image + légende numérotée + descriptif explicatif ---
% Usage : \figemsi{<contenu>}{<légende>}{<label>}{<descriptif>}
\newcommand{\figemsi}[4]{%
  \begin{figure}[H]
    \centering
    #1
    \caption{#2}
    \label{fig:#3}
    \vspace{-2pt}
    \begin{minipage}{0.92\linewidth}
      \small\itshape\color{gristexte}
      \textbf{Descriptif~:} #4
    \end{minipage}
  \end{figure}}

% --- Version prête pour une image déjà présente sur disque ---
% Usage : \figimg{chemin/image}{largeur}{légende}{label}{descriptif}
\newcommand{\figimg}[5]{%
  \figemsi{\includegraphics[width=#2\linewidth]{#1}}{#3}{#4}{#5}}

  \end{minipage}\end{center}}
